-- updated_at no se actualizaba solo — esto lo hace automático en cada UPDATE,
-- sin necesidad de tocar el código Java cada vez que se edite un producto
CREATE OR REPLACE FUNCTION set_updated_at_products()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_products_updated_at ON products;
CREATE TRIGGER trg_products_updated_at
    BEFORE UPDATE ON products
    FOR EACH ROW
    EXECUTE FUNCTION set_updated_at_products();

--  MIGRACIÓN: Folios consecutivos de tickets (INV-000001, INV-000002...)
--  Ejecutar UNA VEZ en PostgreSQL antes de arrancar el backend actualizado.
-- ═══════════════════════════════════════════════════════════════════════
 
-- Secuencia atómica para numerar tickets. Arranca donde va tu contador
-- de ventas para que el folio "cuente" desde tus ventas reales:
-- toma el número de ventas existentes + 1 como punto de partida.
DO $$
DECLARE
    inicio BIGINT;
BEGIN
    SELECT COUNT(*) + 1 INTO inicio FROM sales;
    EXECUTE format(
        'CREATE SEQUENCE IF NOT EXISTS seq_ticket_folio START %s', inicio);
END $$;
 
-- NOTAS:
-- 1. Los tickets ANTERIORES conservan su folio aleatorio (INV-0B39C38D).
--    NO se renumeran a propósito: esos folios ya viajaron al SAT dentro
--    de facturas globales timbradas (campo NoIdentificacion) y deben
--    seguir coincidiendo con tus registros. Cambiarlos rompería la
--    trazabilidad de lo ya facturado.
-- 2. A partir de la primera venta con el sistema actualizado, los folios
--    salen consecutivos: INV-000015, INV-000016, ...
-- 3. No puede haber choques con los folios viejos: los aleatorios tienen
--    8 caracteres hexadecimales y los nuevos 6 dígitos; además la columna
--    invoice_number es UNIQUE, la base lo rechazaría de cualquier forma.
-- 4. Si la secuencia avanza y una venta falla (rollback), puede quedar un
--    "hueco" (ej. INV-000007 no existe). Es normal y fiscalmente
--    irrelevante: el folio del ticket es control interno; lo que importa
--    es que sean únicos y rastreables.
